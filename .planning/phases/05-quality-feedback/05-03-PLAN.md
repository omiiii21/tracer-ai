    ```python
    from tracer_ai.api.schemas import EvalConfigResponse, QueueHealthResponse


    @router.get("/admin/eval-config", response_model=EvalConfigResponse)
    async def get_eval_config() -> EvalConfigResponse:
        """Return the current eval thresholds + judge model + prompt version (D-5.13).

        Single source of truth for the bad-answer-queue threshold; frontend
        reads at mount so the UI filter and the operator-set env var stay
        aligned. Read-only; no DB access; pure Settings + module-constant read.
        """
        # Local import keeps eval/ module optional at admin.py import time —
        # if eval.llm_judge fails to import (missing ANTHROPIC_API_KEY in dev),
        # the rest of /admin/* still mounts. Mirrors PATTERNS.md guidance.
        from tracer_ai.eval.llm_judge import PROMPT_VERSION
        from tracer_ai.config import settings

        return EvalConfigResponse(
            threshold=settings.bad_answer_faithfulness_threshold,
            judge_prompt_version=PROMPT_VERSION,
            judge_model=settings.llm_judge_model,
            calibration_date=settings.calibration_date,
        )


    @router.get("/admin/queue-health", response_model=QueueHealthResponse)
    async def get_queue_health(request: Request) -> QueueHealthResponse:
        """Return live counts for the dashboard 5th KpiCard (FBCK-07).

        queue_size: unresolved thumbs-down feedback rows (uses partial index
        feedback_unresolved_idx from Plan 05-02).
        resolved_this_week: feedback rows resolved in the last 7 days.

        Single-user local; no auth. Pure read-only against the feedback table.
        """
        pool: asyncpg.Pool = request.app.state.db_pool
        async with pool.acquire(timeout=2.0) as conn:
            queue_size = await conn.fetchval(
                "SELECT COUNT(*) FROM feedback "
                "WHERE rating = -1 AND resolved_at IS NULL"
            )
            resolved_this_week = await conn.fetchval(
                "SELECT COUNT(*) FROM feedback "
                "WHERE resolved_at >= NOW() - INTERVAL '7 days'"
            )
        return QueueHealthResponse(
            queue_size=int(queue_size or 0),
            resolved_this_week=int(resolved_this_week or 0),
        )
    ```
