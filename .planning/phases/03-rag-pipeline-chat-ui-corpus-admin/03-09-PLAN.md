       ```ts
       export interface DocSummary { id: string; doc_section: string; source_url: string; chunk_count: number; ingested_at: string; }
       export interface ChunkingConfig { chunk_size: number; overlap: number; }
       export interface CorpusState {
         doc_count: number;
         chunk_count: number;
         embedding_model: string;
         embedding_model_version: string;
         last_indexed_at: string | null;
         docs: DocSummary[];
         chunking_config?: ChunkingConfig;
       }
       export interface IngestResponse { ingest_job_id: string; status: "queued"; }
       export interface IngestStatus {
         ingest_job_id: string;
         status: "queued" | "running" | "succeeded" | "failed";
         started_at: string | null;
         finished_at: string | null;
         docs_processed: number;
         docs_total: number | null;
         chunks_written: number;
         progress: number;
         error?: string;
       }
       export type IngestRequest = { source: "claude-docs" } | { urls: string[] };

       export async function getCorpus(): Promise<CorpusState> {
         const r = await fetch("/admin/corpus");
         if (!r.ok) throw new Error(`getCorpus failed: ${r.status}`);
         return r.json();
       }
       export async function postIngest(req: IngestRequest): Promise<IngestResponse> {
         const r = await fetch("/admin/ingest", { method: "POST", body: JSON.stringify(req), headers: {"Content-Type":"application/json"} });
         if (r.status === 409) throw new Error("Ingest already in progress");
         if (!r.ok) throw new Error(`postIngest failed: ${r.status}`);
         return r.json();
       }
       export async function getIngestStatus(jobId: string): Promise<IngestStatus> {
         const r = await fetch(`/admin/ingest/${jobId}`);
         if (!r.ok) throw new Error(`getIngestStatus failed: ${r.status}`);
         return r.json();
       }
       export async function patchChunkingConfig(cfg: ChunkingConfig): Promise<ChunkingConfig> {
         const r = await fetch("/admin/chunking-config", { method: "PATCH", body: JSON.stringify(cfg), headers: {"Content-Type":"application/json"} });
         if (!r.ok) throw new Error(`patchChunkingConfig failed: ${r.status}`);
         return r.json();
       }
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser <- GET /admin/corpus | Backend response includes user-controlled doc source URLs; rendered as text + as href. |
| browser -> POST /admin/ingest | URL list user-controlled; server-side schema is authoritative (Plan 01). |
| browser -> PATCH /admin/chunking-config | Numeric inputs; server-side schema enforces 100..4000 / 0..500. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-09-01 | Tampering | UrlIngestForm client-side regex bypass | mitigate | Server-side Pydantic schema (Plan 01) re-validates each URL with the same `^https?://` regex; client-side check is UX-only. |
| T-03-09-02 | Information Disclosure | DocList source_url rendering | accept | Local-dev only; the operator IS the audience. Source URL truncation prevents accidental layout leaks but isn't a security boundary. |
| T-03-09-03 | Spoofing | /admin/* unauthenticated | accept | Inherited from Plan 07 — v1 single-user local-dev only (ADR 009). |
| T-03-09-04 | Denial of Service | ReindexButton spam-click | mitigate | Two-tap confirm pattern (3s timeout) prevents accidental triggers; server-side 409 (Plan 07) prevents concurrent ingests; mutation disabled during running state. |
| T-03-09-05 | Tampering | ChunkingConfigForm out-of-range value | mitigate | Input min/max attributes enforce client-side; server-side schema enforces authoritatively (Plan 01); chunker constructor (Plan 02) re-validates so even direct DB poisoning cannot produce a working chunker with bad params. |
| T-03-09-06 | Information Disclosure | error toast leakage | mitigate | Error toast surfaces err.message only (e.g. "postIngest failed: 409"); never the full backend traceback. |
| T-03-09-07 | Repudiation | admin actions audit trail | mitigate | Every backend mutation logs structured events (Plan 07); frontend toasts confirm success/failure to operator. |
</threat_model>
