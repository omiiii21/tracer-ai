       export interface ChatRequest { question: string; thread_id?: string; }

       export interface ChatFinal { trace_id: string; cited_chunks: Citation[]; latency_ms: number; input_tokens: number; output_tokens: number; estimated_cost_usd: number; }
       export type SSEEvent = { event: "token"; data: { text: string } } | { event: "final"; data: ChatFinal } | { event: "error"; data: { message: string } };
       export interface FeedbackRequest { trace_id: string; rating: 1 | -1; comment?: string | null; }

       export async function* postChat(req: ChatRequest, signal?: AbortSignal) {
         yield* sseStream("/chat", { method: "POST", body: JSON.stringify(req), headers: { "Content-Type": "application/json" }, signal });
       }

       export async function postFeedback(req: FeedbackRequest): Promise<{id: string; created_at: string}> {
         const res = await fetch("/feedback", { method: "POST", body: JSON.stringify(req), headers: { "Content-Type": "application/json" } });
         if (!res.ok) throw new Error(`postFeedback failed: ${res.status}`);
         return res.json();
       }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser <- POST /chat (SSE) | Token text from the LLM is rendered as DOM text. React's default escaping is the XSS defense. |
| browser <- chunk content in CitationAccordion | Chunk content is also rendered as text (font-mono + whitespace-pre-wrap). |
| user input -> Textarea -> POST /chat | User-controlled input; backend validates via Pydantic (Plan 06 max_length=4000). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-08-01 | Tampering | Chat.tsx token rendering | mitigate | React's text rendering escapes by default. NEVER use dangerouslySetInnerHTML. The cursor character `▋` is a literal Unicode character, not HTML. |
| T-03-08-02 | Tampering | Citation.tsx chunk content | mitigate | Same — chunk content rendered inside <pre> as text; no innerHTML; no dangerouslySetInnerHTML. |
| T-03-08-03 | Information Disclosure | Source URL click-through | accept | source_url displayed as clickable link via target=_blank rel=noreferrer. Phishing risk is the operator clicking a malicious URL ingested into the corpus — local-dev only, operator IS the user. |
| T-03-08-04 | Spoofing | trace_id forgery in URL | accept | Local-dev only; trace_id URLs go to TraceStub (Phase 3) which does not query the DB. Phase 4 trace explorer would need its own ACL boundary. |
| T-03-08-05 | Denial of Service | abandoned SSE connection | mitigate | AbortController plumbed through postChat signal; component cleanup (useEffect return) cancels in-flight requests on unmount. |
| T-03-08-06 | Information Disclosure | comment dialog (PII) | accept | Comment is user-supplied; saved verbatim. Local-dev only. |
| T-03-08-07 | Tampering | Pin gate erosion | mitigate | After `npm install` of date-fns + shadcn-added Radix deps, re-grep for react@^19 and tailwindcss@^4. Both must be 0. Pre-commit gate from Phase 2 enforces. |
